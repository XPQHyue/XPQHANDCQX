给零基础同学： GPIO就是芯片的"引脚"，这章学习如何用程序控制引脚的电平。

GPIO初始化代码
位置：main.c的main函数开头

#include "stm32f10x.h"

// GPIO初始化函数
void GPIO_Configuration(void)
{
    GPIO_InitTypeDef GPIO_InitStructure;
    
    // 1. 开启GPIOB时钟
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOB, ENABLE);
    
    // 2. 配置PB0为推挽输出（控制LED）
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_0;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_Out_PP;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOB, &GPIO_InitStructure);
    
    // 3. 配置PA0为上拉输入（连接按键）
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA, ENABLE);
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_0;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IPU;  // 上拉输入模式
    GPIO_Init(GPIOA, &GPIO_InitStructure);
}

// main函数中的调用位置
int main(void)
{
    GPIO_Configuration();  // 放在main函数开头
    
    while(1)
    {
        // 你的主循环代码
    }
}

第2章：中断系统（外部中断）

观看P7-P9

给零基础同学： 中断就是"打断"，当有紧急事件发生时，CPU会暂停当前工作去处理。

外部中断配置
位置：新建exti.c文件或放在main.c开头

#include "stm32f10x.h"
#include "stm32f10x_exti.h"

// 外部中断初始化函数
void EXTI_Configuration(void)
{
    EXTI_InitTypeDef EXTI_InitStructure;
    GPIO_InitTypeDef GPIO_InitStructure;
    NVIC_InitTypeDef NVIC_InitStructure;
    
    // 1. 开启GPIOA时钟
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA, ENABLE);
    
    // 2. 配置PA0为上拉输入（连接按键）
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_0;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IPU;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOA, &GPIO_InitStructure);
    
    // 3. 开启AFIO时钟（必须开启！）
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_AFIO, ENABLE);
    
    // 4. 连接GPIO到EXTI线
    GPIO_EXTILineConfig(GPIO_PortSourceGPIOA, GPIO_PinSource0);
    
    // 5. 配置EXTI线0
    EXTI_InitStructure.EXTI_Line = EXTI_Line0;
    EXTI_InitStructure.EXTI_Mode = EXTI_Mode_Interrupt;
    EXTI_InitStructure.EXTI_Trigger = EXTI_Trigger_Falling;  // 下降沿触发
    EXTI_InitStructure.EXTI_LineCmd = ENABLE;
    EXTI_Init(&EXTI_InitStructure);
    
    // 6. 配置NVIC中断优先级
    NVIC_InitStructure.NVIC_IRQChannel = EXTI0_IRQn;
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 0;
    NVIC_InitStructure.NVIC_IRQChannelSubPriority = 0;
    NVIC_InitStructure.NVIC_IRQChannelCmd = ENABLE;
    NVIC_Init(&NVIC_InitStructure);
}

// 中断服务函数（必须放在stm32f10x_it.c文件中）
void EXTI0_IRQHandler(void)
{
    if(EXTI_GetITStatus(EXTI_Line0) != RESET)
    {
        // 按键按下时执行的操作
        GPIO_WriteBit(GPIOB, GPIO_Pin_0, 
            (BitAction)(1 - GPIO_ReadOutputDataBit(GPIOB, GPIO_Pin_0)));
        
        // 清除中断标志位（必须清除！）
        EXTI_ClearITPendingBit(EXTI_Line0);
    }
}

重要！代码放置位置说明

EXTI_Configuration()函数 → 放在main.c的main函数开头调用

EXTI0_IRQHandler()函数 → 必须放在stm32f10x_it.c文件中！

stm32f10x_it.c位置：工程目录 → User文件夹 → stm32f10x_it.c

提示： 找不到这个文件？在Keil中按F12或右键工程查看文件结构。

第3章：PWM输出（控制LED亮度）

观看P10-P12

给零基础同学： PWM就是"脉冲宽度调制"，可以控制LED的亮度、电机的速度。

PWM配置（定时器2通道1）
位置：新建pwm.c文件或放在main.c开头

#include "stm32f10x.h"
#include "stm32f10x_tim.h"

// PWM初始化函数（使用TIM2通道1，对应PA0引脚）
void PWM_Configuration(void)
{
    GPIO_InitTypeDef GPIO_InitStructure;
    TIM_TimeBaseInitTypeDef TIM_TimeBaseStructure;
    TIM_OCInitTypeDef TIM_OCInitStructure;
    
    // 1. 开启时钟
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA, ENABLE);
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_TIM2, ENABLE);
    
    // 2. 配置PA0为复用推挽输出（PWM输出必须用复用功能！）
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_0;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_PP;  // 复用推挽输出
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOA, &GPIO_InitStructure);
    
    // 3. 配置定时器2时基
    TIM_TimeBaseStructure.TIM_Period = 999;          // 自动重装载值：1000-1
    TIM_TimeBaseStructure.TIM_Prescaler = 71;        // 预分频值：72-1
    TIM_TimeBaseStructure.TIM_ClockDivision = 0;
    TIM_TimeBaseStructure.TIM_CounterMode = TIM_CounterMode_Up;
    TIM_TimeBaseInit(TIM2, &TIM_TimeBaseStructure);
    
    // 4. 配置PWM输出通道1
    TIM_OCInitStructure.TIM_OCMode = TIM_OCMode_PWM1;
    TIM_OCInitStructure.TIM_OutputState = TIM_OutputState_Enable;
    TIM_OCInitStructure.TIM_OCPolarity = TIM_OCPolarity_High;
    TIM_OCInitStructure.TIM_Pulse = 500;  // 初始占空比50%（500/1000）
    TIM_OC1Init(TIM2, &TIM_OCInitStructure);
    TIM_OC1PreloadConfig(TIM2, TIM_OCPreload_Enable);
    
    // 5. 开启定时器预装载和定时器
    TIM_ARRPreloadConfig(TIM2, ENABLE);
    TIM_Cmd(TIM2, ENABLE);
}

// 修改PWM占空比的函数
void PWM_SetDutyCycle(uint16_t duty)  // duty: 0-1000
{
    if(duty > 999) duty = 999;
    TIM_SetCompare1(TIM2, duty);
}

// main函数中使用示例
int main(void)
{
    uint16_t brightness = 0;
    
    PWM_Configuration();  // 初始化PWM
    
    while(1)
    {
        // 呼吸灯效果
        for(brightness = 0; brightness < 1000; brightness += 10)
        {
            PWM_SetDutyCycle(brightness);
            Delay(10000);  // 简单延时函数
        }
        for(brightness = 1000; brightness > 0; brightness -= 10)
        {
            PWM_SetDutyCycle(brightness);
            Delay(10000);
        }
    }
}

第4章：I2C通信与OLED显示

观看P16-P18

给零基础同学： I2C只需要两根线（SDA数据线、SCL时钟线）就能连接多个设备，OLED就是小屏幕。

I2C初始化（模拟I2C）
位置：新建oled.c和oled.h文件

#define __OLED_H

#include "stm32f10x.h"

void OLED_Init(void);
void OLED_Clear(void);
void OLED_ShowString(uint8_t line, uint8_t column, char *str);
void OLED_ShowNum(uint8_t line, uint8_t column, uint32_t num, uint8_t len);
void OLED_ShowChinese(uint8_t x, uint8_t y, uint8_t no);

#endif

// oled.c文件部分内容
#include "oled.h"
#include "oledfont.h"  // 字库文件

// 引脚定义（使用PB8-PB9做模拟I2C）
#define OLED_SCL_PIN    GPIO_Pin_8
#define OLED_SDA_PIN    GPIO_Pin_9
#define OLED_SCL_PORT   GPIOB
#define OLED_SDA_PORT   GPIOB

// I2C起始信号
void OLED_I2C_Start(void)
{
    OLED_SDA(1);
    OLED_SCL(1);
    OLED_SDA(0);
    OLED_SCL(0);
}

// I2C停止信号
void OLED_I2C_Stop(void)
{
    OLED_SDA(0);
    OLED_SCL(1);
    OLED_SDA(1);
}

// OLED初始化函数
void OLED_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStructure;
    
    // 开启GPIOB时钟
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOB, ENABLE);
    
    // 配置PB8、PB9为开漏输出
    GPIO_InitStructure.GPIO_Pin = OLED_SCL_PIN | OLED_SDA_PIN;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_Out_OD;  // 开漏输出
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOB, &GPIO_InitStructure);
    
    // OLED硬件初始化序列
    OLED_WriteCommand(0xAE);  // 关闭显示
    OLED_WriteCommand(0xD5);  // 设置显示时钟分频比/振荡器频率
    OLED_WriteCommand(0x80);
    OLED_WriteCommand(0xA8);  // 设置多路复用率
    OLED_WriteCommand(0x3F);
    // ... 更多初始化命令
    
    OLED_Clear();  // 清屏
    OLED_WriteCommand(0xAF);  // 开启显示
}

// 显示字符串函数
void OLED_ShowString(uint8_t line, uint8_t column, char *str)
{
    uint8_t i;
    for(i = 0; str[i] != '\0'; i++)
    {
        OLED_ShowChar(line, column + i, str[i]);
    }
}

// main.c中使用示例
int main(void)
{
    OLED_Init();  // 初始化OLED
    
    OLED_ShowString(1, 1, "Hello STM32!");
    OLED_ShowString(2, 1, "OLED Display Test");
    OLED_ShowString(3, 1, "Temp: 25.6C");
    OLED_ShowString(4, 1, "Humidity: 60%");
    
    while(1)
    {
        // 主循环
    }
}

// usart.c文件
#include "usart.h"

// 串口1初始化函数
void USART1_Config(uint32_t baudrate)
{
    GPIO_InitTypeDef GPIO_InitStructure;
    USART_InitTypeDef USART_InitStructure;
    
    // 1. 开启时钟（USART1在APB2总线上）
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_USART1 | 
                          RCC_APB2Periph_GPIOA, ENABLE);
    
    // 2. 配置PA9为复用推挽输出（TX发送引脚）
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_9;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_PP;  // 复用推挽输出
    GPIO_Init(GPIOA, &GPIO_InitStructure);
    
    // 3. 配置PA10为浮空输入（RX接收引脚）
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_10;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IN_FLOATING;  // 浮空输入
    GPIO_Init(GPIOA, &GPIO_InitStructure);
    
    // 4. 配置USART1参数
    USART_InitStructure.USART_BaudRate = baudrate;  // 波特率
    USART_InitStructure.USART_WordLength = USART_WordLength_8b;
    USART_InitStructure.USART_StopBits = USART_StopBits_1;
    USART_InitStructure.USART_Parity = USART_Parity_No;
    USART_InitStructure.USART_HardwareFlowControl = USART_HardwareFlowControl_None;
    USART_InitStructure.USART_Mode = USART_Mode_Rx | USART_Mode_Tx;
    
    USART_Init(USART1, &USART_InitStructure);
    USART_Cmd(USART1, ENABLE);
}

// 发送一个字节
void USART1_SendByte(uint8_t byte)
{
    USART_SendData(USART1, byte);
    while(USART_GetFlagStatus(USART1, USART_FLAG_TXE) == RESET);
}

// 发送字符串
void USART1_SendString(char *str)
{
    while(*str)
    {
        USART1_SendByte(*str++);
    }
}

// 重定向printf到串口（重要！）
int fputc(int ch, FILE *f)
{
    USART_SendData(USART1, (uint8_t)ch);
    while(USART_GetFlagStatus(USART1, USART_FLAG_TXE) == RESET);
    return ch;
}

// main.c中使用示例
int main(void)
{
    uint8_t received_data;
    
    USART1_Config(115200);  // 初始化串口，波特率115200
    
    printf("STM32 USART Test Start!\r\n");
    printf("System Clock: %d Hz\r\n", SystemCoreClock);
    
    while(1)
    {
        // 接收数据并回传（串口助手发送什么就回传什么）
        if(USART_GetFlagStatus(USART1, USART_FLAG_RXNE) != RESET)
        {
            received_data = USART_ReceiveData(USART1);
            printf("Received: %c (0x%02X)\r\n", received_data, received_data);
            
            // 回传数据
            USART1_SendByte(received_data);
        }
    }
}

第6章：内部Flash读写

观看P19-P21

给零基础同学： Flash就是STM32内部的"硬盘"，可以保存数据，断电后数据不会丢失。

内部Flash操作
位置：新建flash.c和flash.h文件

// flash.h头文件
#ifndef __FLASH_H
#define __FLASH_H

#include "stm32f10x.h"

// Flash起始地址（根据STM32F103C8T6的64KB Flash）
#define FLASH_START_ADDR  0x08000000
#define FLASH_SIZE        0x10000      // 64KB
#define FLASH_PAGE_SIZE   0x400        // 1KB per page

// 使用最后一页存储用户数据（避免擦除程序）
#define USER_DATA_ADDR    (FLASH_START_ADDR + FLASH_SIZE - FLASH_PAGE_SIZE)

void Flash_ErasePage(uint32_t page_address);
void Flash_WriteData(uint32_t address, uint32_t *data, uint16_t length);
void Flash_ReadData(uint32_t address, uint32_t *data, uint16_t length);

#endif

// flash.c文件
#include "flash.h"

// 解锁Flash
static void Flash_Unlock(void)
{
    FLASH_Unlock();
    FLASH_ClearFlag(FLASH_FLAG_BSY | FLASH_FLAG_PGERR | FLASH_FLAG_WRPRTERR);
}

// 锁定Flash
static void Flash_Lock(void)
{
    FLASH_Lock();
}

// 擦除指定页
void Flash_ErasePage(uint32_t page_address)
{
    Flash_Unlock();
    
    // 等待Flash不忙
    while(FLASH_GetStatus() != FLASH_COMPLETE);
    
    // 擦除页
    FLASH_ErasePage(page_address);
    
    // 等待擦除完成
    while(FLASH_GetStatus() != FLASH_COMPLETE);
    
    Flash_Lock();
}

// 写入数据到Flash
void Flash_WriteData(uint32_t address, uint32_t *data, uint16_t length)
{
    uint16_t i;
    
    Flash_Unlock();
    
    for(i = 0; i < length; i++)
    {
        // 写入一个字（32位）
        FLASH_ProgramWord(address + i * 4, data[i]);
        
        // 等待写入完成
        while(FLASH_GetStatus() != FLASH_COMPLETE);
        
        // 验证写入的数据
        if(*(uint32_t*)(address + i * 4) != data[i])
        {
            // 写入失败处理
            break;
        }
    }
    
    Flash_Lock();
}

// 从Flash读取数据
void Flash_ReadData(uint32_t address, uint32_t *data, uint16_t length)
{
    uint16_t i;
    
    for(i = 0; i < length; i++)
    {
        data[i] = *(uint32_t*)(address + i * 4);
    }
}

// main.c中使用示例
int main(void)
{
    uint32_t write_data[4] = {0x12345678, 0x87654321, 0xABCDEF01, 0xFEDCBA09};
    uint32_t read_data[4];
    
    printf("Flash Test Start!\r\n");
    
    // 1. 擦除最后一页Flash
    Flash_ErasePage(USER_DATA_ADDR);
    printf("Flash page erased.\r\n");
    
    // 2. 写入数据
    Flash_WriteData(USER_DATA_ADDR, write_data, 4);
    printf("Data written to Flash.\r\n");
    
    // 3. 读取数据
    Flash_ReadData(USER_DATA_ADDR, read_data, 4);
    printf("Data read from Flash:\r\n");
    
    for(int i = 0; i < 4; i++)
    {
        printf("  Address 0x%08X: 0x%08X\r\n", 
               USER_DATA_ADDR + i * 4, read_data[i]);
    }
    
    // 4. 验证数据
    uint8_t verify_ok = 1;
    for(int i = 0; i < 4; i++)
    {
        if(write_data[i] != read_data[i])
        {
            verify_ok = 0;
            break;
        }
    }
    
    if(verify_ok)
        printf("Flash verify: PASS!\r\n");
    else
        printf("Flash verify: FAIL!\r\n");
    
    while(1);
}
